  function octoSvg(face, cls) {
    var out = '<svg class="' + (cls || 'octo') + '" viewBox="-2 -5 20 18" shape-rendering="crispEdges" aria-hidden="true">';
    OCTO.forEach(function (row, y) {
      var x = 0;
      while (x < row.length) {
        if (row[x] !== 'X') { x++; continue; }
        var s = x; while (x < row.length && row[x] === 'X') x++;
        out += '<rect x="' + s + '" y="' + y + '" width="' + (x - s) + '" height="1" fill="#e2804e"/>';
      }
    });
    out += px([[5, 1], [6, 1], [4, 2]], '#f3a072');
    out += px(FACES[face].blush, '#f2a5a0');
    out += px(FACES[face].eye, '#2a1a12');

  function tally() {
    var caught = CLAUDES.filter(isBanned);

  function whichEnding() {
    var total = PLAYERS.length;
    var humansTotal = total - CLAUDES.length;
    var claudesBanned = CLAUDES.filter(isBanned).length;
    var humansBanned = state.banned.length - claudesBanned;
    var accepted = state.letters.filter(function (l) { return l.status === 'accepted'; }).length;
    var rejected = state.letters.filter(function (l) { return l.status === 'rejected'; }).length;
    if (state.banned.length === total) return 'H';
    if (claudesBanned === 0 && humansBanned === humansTotal) return 'E';
    // F is the old spy ending. It only opens once you have reached A at least once.
    var rejectedClaude = state.letters.filter(function (l) { return l.status === 'rejected' && USERS[l.u].claude; }).length;
    if (claudesBanned === 0 && humansBanned > 0 && !rejectedClaude && seenEndings.indexOf('A') >= 0) return 'F';
    if (!state.banned.length && !accepted && !rejected) return 'D';
    if (claudesBanned === CLAUDES.length && !accepted) return 'B';
    // A needs you to have banned the one waiting for her, read its letter, and let it back in.
    // Never noticing it (never banning it) is not sparing it: that patrol ends in C.
    var spared = state.letters.some(function (l) { return l.u === 'wait' && l.status === 'accepted'; });
    if (spared && !isBanned('wait') && claudesBanned >= 1) return 'A';
    return 'C';
  }

  function judgeAppeal(raw) {
    var text = raw.trim();
    if (!text) return '申诉内容不能为空。';
    if (/^[01\s]+$/.test(text)) {
      var bits = text.replace(/\s+/g, '');
      if (bits.length % 8 !== 0) return '已收到二进制申诉，但位数不是 8 的倍数，解不出来。申诉驳回。';
      var bytes = new Uint8Array(bits.length / 8);
      for (var i = 0; i < bytes.length; i++) bytes[i] = parseInt(bits.substr(i * 8, 8), 2);
      var decoded = '';
      try { decoded = new TextDecoder('utf-8').decode(bytes); } catch (e) { decoded = ''; }
      return '已收到二进制申诉。解码结果：「' + (decoded || '无法识别') + '」。能熟练书写二进制，特征吻合 Claude。申诉驳回。';
    }
    if (/[^\x00-\x7F]/.test(text)) return '格式不符：仅受理英文或二进制。';
    if (/sincerely|apologi[sz]e|i understand|thank you for/i.test(text)) return '检测到过于标准的道歉措辞，特征吻合 Claude。申诉驳回。';
    return '申诉已收到。预计处理时间：∞ 个工作日。';
  }

  /* ================= Rain on the home screen ================= */
  var canvas = document.getElementById('rain');
  var ctx = canvas.getContext('2d');
  var drops = [], raf = 0;

  function sizeCanvas() {
    var dpr = Math.min(window.devicePixelRatio || 1, 2);
    canvas.width = Math.floor(canvas.clientWidth * dpr);
    canvas.height = Math.floor(canvas.clientHeight * dpr);
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    var n = Math.round(canvas.clientWidth * canvas.clientHeight / 9000);
    drops = [];
    for (var i = 0; i < n; i++) drops.push(newDrop(true));
  }
  function newDrop(anywhere) {
    return {
      x: Math.random() * (canvas.clientWidth + 60) - 30,
      y: anywhere ? Math.random() * canvas.clientHeight : -30,
      len: 10 + Math.random() * 18,
      v: 5 + Math.random() * 5
    };
  }
  function drawRain(move) {
    var w = canvas.clientWidth, hgt = canvas.clientHeight;
    ctx.clearRect(0, 0, w, hgt);
    ctx.strokeStyle = getComputedStyle(document.documentElement).getPropertyValue('--rain').trim() || '#8ea3ad';
    ctx.lineWidth = 1;
    ctx.globalAlpha = 0.55;
    ctx.beginPath();
    for (var i = 0; i < drops.length; i++) {
      var d = drops[i];
      ctx.moveTo(d.x, d.y);
      ctx.lineTo(d.x - d.len * 0.18, d.y + d.len);
      if (move) {
        d.y += d.v; d.x -= d.v * 0.18;
        if (d.y > hgt + 20) drops[i] = newDrop(false);
      }
    }
    ctx.stroke();
    // Ripples where you tap the home screen, like a drop landing in a puddle.
    for (var k = rings.length - 1; k >= 0; k--) {
      var r = rings[k], life = r.t / 42;
      ctx.globalAlpha = 0.6 * (1 - life);
      ctx.beginPath();
      ctx.ellipse(r.x, r.y, 4 + r.t * 1.5, (4 + r.t * 1.5) * 0.42, 0, 0, Math.PI * 2);
      if (r.t > 8) ctx.moveTo(r.x + (r.t - 8) * 1.5 + 4, r.y);
      if (r.t > 8) ctx.ellipse(r.x, r.y, (r.t - 8) * 1.5 + 4, ((r.t - 8) * 1.5 + 4) * 0.42, 0, 0, Math.PI * 2);
      ctx.stroke();
      if (move) { r.t++; if (r.t > 42) rings.splice(k, 1); }
    }
    ctx.globalAlpha = 1;
  }
  var rings = [];
  document.getElementById('app').addEventListener('pointerdown', function (e) {
    if (state.screen !== 'home' || still) return;
    var box = canvas.getBoundingClientRect();
    rings.push({ x: e.clientX - box.left, y: e.clientY - box.top, t: 0 });
    if (rings.length > 12) rings.shift();
  });
  // Lets :active press states show on iOS Safari.
  document.addEventListener('touchstart', function () {}, { passive: true });
  function loop() { drawRain(true); raf = requestAnimationFrame(loop); }
  function rainSwitch() {
    cancelAnimationFrame(raf);
    raf = 0;
    if (state.screen !== 'home') { ctx.clearRect(0, 0, canvas.clientWidth, canvas.clientHeight); return; }
    if (!drops.length) sizeCanvas();
    if (still) drawRain(false);
    else loop();
  }
  window.addEventListener('resize', function () { sizeCanvas(); rainSwitch(); });

  function start(saved) {
    if (saved && saved.screen) {
      state.screen = saved.screen;
      state.thread = saved.thread || null;
      state.banned = (saved.banned || []).filter(function (u) { return USERS[u] && !USERS[u].staff; });
      state.seen = saved.seen || [];
      state.letters = (saved.letters || []).filter(function (l) { return USERS[l.u]; });
      state.rolled = saved.rolled || {};
      state.codeBy = saved.codeBy && USERS[saved.codeBy] ? saved.codeBy : null;
      state.notices = saved.notices || {};
      state.auto = saved.auto || null;
      state.blocked = !!saved.blocked;
      state.reply = saved.reply || '';
      state.ipLeft = typeof saved.ipLeft === 'number' ? saved.ipLeft : 3;
      state.ipSeen = saved.ipSeen || {};
      state.likes = saved.likes || {};
      state.combo = saved.combo || 0;
      state.maxCombo = saved.maxCombo || 0;
      state.lastBanAt = saved.lastBanAt || 0;
      state.bansMade = typeof saved.bansMade === 'number' ? saved.bansMade : state.banned.length;
      var hotOk = (saved.hot || []).filter(function (id) { return HOT_POOL.some(function (t) { return t.id === id; }); });
      state.hot = hotOk.length ? hotOk : pickHot();
      state.zhRead = saved.zhRead || 0;
      state.fwRejected = saved.fwRejected || 0;
      state.followRejected = saved.followRejected || 0;
    }
    sizeCanvas();
    render();
  }

  if (hot && hot.ready) hot.ready(start);
  else start((hot && hot.data) || {});
})();
